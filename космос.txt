        const url = safeTrim(urlString);

      function isProbablyUrl(urlString){
        const url = safeTrim(urlString);
        if (!url) return false;
        return /^(https?:\/\/|mailto:|tel:)/i.test(url) || /^[a-z0-9.-]+\.[a-z]{2,}([\/?#].*)?$/i.test(url);
      }

      function makeHref(urlString){
        const url = normalizeUrl(urlString);
        if (!url) return "";
        if (/^(https?:\/\/|mailto:|tel:)/i.test(url)) return url;
        if (/^[a-z0-9.-]+\.[a-z]{2,}([\/?#].*)?$/i.test(url)) return "https://" + url;
        return url;

      // Starfield + nebula canvas (parallax layers)
      const starfield = (function(){
        const canvas = document.getElementById("starfield");
        const context = canvas.getContext("2d", { alpha: true, desynchronized: true });

        let width = 0;
        let height = 0;
        let devicePixelRatio = 1;

        let running = false;
        let animationFrameId = 0;
        let lastTime = 0;

        const pointer = { x: 0, y: 0, has: false };

        const layers = [
          { count: 120, speed: 0.020, sizeMin: 0.6, sizeMax: 1.6, alpha: 0.70, tint: [0, 255, 247] },
          { count: 90, speed: 0.040, sizeMin: 0.9, sizeMax: 2.2, alpha: 0.60, tint: [255, 3, 218] },
          { count: 60, speed: 0.070, sizeMin: 1.1, sizeMax: 2.8, alpha: 0.55, tint: [255, 255, 255] }
        ];

        let stars = [];
        let nebulaBlobs = [];

        function randomBetween(min, max){
          return min + Math.random() * (max - min);
        }

        function resize(){
          const rect = canvas.getBoundingClientRect();
          devicePixelRatio = Math.min(2, window.devicePixelRatio || 1);
          width = Math.max(2, Math.floor(rect.width * devicePixelRatio));
          height = Math.max(2, Math.floor(rect.height * devicePixelRatio));

          canvas.width = width;
          canvas.height = height;

          context.setTransform(1,0,0,1,0,0);
          context.imageSmoothingEnabled = true;

          buildStars();
          buildNebula();
          draw(0);
        }

        function buildStars(){
          stars = layers.map((layer, layerIndex) => {
            const list = [];
            for (let i = 0; i < layer.count; i++){
              list.push({
                x: Math.random(),
                y: Math.random(),
                radius: randomBetween(layer.sizeMin, layer.sizeMax) * devicePixelRatio,
                alpha: randomBetween(0.35, 1.0),
                twinkle: randomBetween(0.6, 1.6),
                phase: randomBetween(0, Math.PI * 2),
                layerIndex
              });
            }
            return list;
          });
        }

        function buildNebula(){
          const blobCount = 7;
          nebulaBlobs = [];
          for (let i = 0; i < blobCount; i++){
            nebulaBlobs.push({
              x: Math.random(),
              y: Math.random(),
              radius: randomBetween(0.18, 0.42),
              driftX: randomBetween(-0.015, 0.015),
              driftY: randomBetween(-0.010, 0.010),
              hue: i % 3 === 0 ? "cyan" : (i % 3 === 1 ? "magenta" : "orange"),
              alpha: randomBetween(0.08, 0.16)
            });
          }
        }

        function clear(){
          context.clearRect(0, 0, width, height);
        }

        function drawNebula(deltaSeconds){
          const glow = Number(getComputedStyle(document.documentElement).getPropertyValue("--glow")) || 0.85;
          context.save();

          // very soft darkening vignette for depth
          const vignette = context.createRadialGradient(
            width * 0.5, height * 0.5, Math.min(width, height) * 0.08,
            width * 0.5, height * 0.5, Math.min(width, height) * 0.62
          );
          vignette.addColorStop(0, "rgba(0,0,0,0.00)");
          vignette.addColorStop(1, "rgba(0,0,0,0.42)");
          context.fillStyle = vignette;
          context.fillRect(0,0,width,height);

          // nebula blobs
          context.globalCompositeOperation = "screen";
          for (const blob of nebulaBlobs){
            blob.x += blob.driftX * deltaSeconds;
            blob.y += blob.driftY * deltaSeconds;

            if (blob.x < -0.2) blob.x = 1.2;
            if (blob.x > 1.2) blob.x = -0.2;
            if (blob.y < -0.2) blob.y = 1.2;
            if (blob.y > 1.2) blob.y = -0.2;

            const px = blob.x * width + (pointer.has ? (pointer.x - 0.5) * width * 0.02 : 0);
            const py = blob.y * height + (pointer.has ? (pointer.y - 0.5) * height * 0.02 : 0);
            const r = blob.radius * Math.min(width, height);

            const gradient = context.createRadialGradient(px, py, r * 0.08, px, py, r);

            let color = { r: 0, g: 0, b: 0 };
            if (blob.hue === "cyan") color = { r: 0, g: 255, b: 247 };
            if (blob.hue === "magenta") color = { r: 255, g: 3, b: 218 };
            if (blob.hue === "orange") color = { r: 255, g: 77, b: 0 };

            const a = blob.alpha * (0.65 + 0.55 * glow);
            gradient.addColorStop(0, "rgba(" + color.r + "," + color.g + "," + color.b + "," + (a * 1.00) + ")");
            gradient.addColorStop(0.55, "rgba(" + color.r + "," + color.g + "," + color.b + "," + (a * 0.20) + ")");
            gradient.addColorStop(1, "rgba(" + color.r + "," + color.g + "," + color.b + ",0)");

            context.fillStyle = gradient;
            context.beginPath();
            context.arc(px, py, r, 0, Math.PI * 2);
            context.fill();
          }

          context.restore();
        }

        function drawStars(timeSeconds){
          const glow = Number(getComputedStyle(document.documentElement).getPropertyValue("--glow")) || 0.85;

          context.save();
          context.globalCompositeOperation = "lighter";

          for (let layerIndex = 0; layerIndex < layers.length; layerIndex++){
            const layer = layers[layerIndex];
            const list = stars[layerIndex];

            const parallaxX = pointer.has ? (pointer.x - 0.5) * width * layer.speed * 1.6 : 0;
            const parallaxY = pointer.has ? (pointer.y - 0.5) * height * layer.speed * 1.6 : 0;

            for (const star of list){
              const twinkle = 0.55 + 0.45 * Math.sin(timeSeconds * star.twinkle + star.phase);
              const alpha = clamp(layer.alpha * star.alpha * twinkle * (0.70 + glow * 0.35), 0, 1);

              const x = (star.x * width + parallaxX + timeSeconds * layer.speed * 260 * devicePixelRatio) % width;
              const y = (star.y * height + parallaxY + timeSeconds * layer.speed * 120 * devicePixelRatio) % height;

              const tint = layer.tint;
              context.fillStyle = "rgba(" + tint[0] + "," + tint[1] + "," + tint[2] + "," + alpha + ")";

              context.beginPath();
              context.arc(x, y, star.radius, 0, Math.PI * 2);
              context.fill();

              // tiny cross sparkle occasionally for depth (very cheap)
              if (star.radius > 1.4 * devicePixelRatio && twinkle > 0.86){
                context.globalAlpha = alpha * 0.85;
                context.fillRect(x - 3 * devicePixelRatio, y, 6 * devicePixelRatio, 1 * devicePixelRatio);
                context.fillRect(x, y - 3 * devicePixelRatio, 1 * devicePixelRatio, 6 * devicePixelRatio);
                context.globalAlpha = 1;
              }
            }
          }

          context.restore();
        }

        function draw(deltaSeconds){
          const timeSeconds = performance.now() / 1000;
          clear();
          drawNebula(deltaSeconds);
          drawStars(timeSeconds);

          // glassy inner ring edge hint
          context.save();
          context.globalCompositeOperation = "source-over";
          const edgeGradient = context.createRadialGradient(width * 0.5, height * 0.5, Math.min(width, height) * 0.37, width * 0.5, height * 0.5, Math.min(width, height) * 0.51);
          edgeGradient.addColorStop(0, "rgba(255,255,255,0)");
          edgeGradient.addColorStop(0.7, "rgba(0,255,247,0.06)");
          edgeGradient.addColorStop(1, "rgba(0,0,0,0.40)");
          context.fillStyle = edgeGradient;
          context.fillRect(0, 0, width, height);
          context.restore();
        }

        function tick(now){
          if (!running) return;
          if (!lastTime) lastTime = now;
          const deltaMilliseconds = now - lastTime;
          lastTime = now;
          const deltaSeconds = clamp(deltaMilliseconds / 1000, 0, 0.05);

          draw(deltaSeconds);
          animationFrameId = window.requestAnimationFrame(tick);
        }

        function start(){
          if (running) return;
          running = true;
          lastTime = 0;
          animationFrameId = window.requestAnimationFrame(tick);
        }

        function stop(drawOnce){
          running = false;
          if (animationFrameId) window.cancelAnimationFrame(animationFrameId);
          animationFrameId = 0;
          lastTime = 0;
          if (drawOnce) draw(0);
        }

        function setPointerFromEvent(event){
          const rect = canvas.getBoundingClientRect();
          const x = (event.clientX - rect.left) / rect.width;
          const y = (event.clientY - rect.top) / rect.height;
          pointer.x = clamp(x, 0, 1);
          pointer.y = clamp(y, 0, 1);
          pointer.has = true;
        }

        function clearPointer(){
          pointer.has = false;
        }

        window.addEventListener("resize", () => resize(), { passive: true });

        const porthole = document.getElementById("porthole");
        porthole.addEventListener("pointermove", (event) => setPointerFromEvent(event), { passive: true });
        porthole.addEventListener("pointerleave", () => clearPointer(), { passive: true });

        resize();

        return { start, stop, resize };
      })();
