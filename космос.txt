  <script>
    (function () {
      "use strict";

      /* ----------------------------
         DOM + controls
      ---------------------------- */
      const canvas = document.getElementById("spaceCanvas");
      const context = canvas.getContext("2d", { alpha: false });

      const warpButton = document.getElementById("warpButton");
      const namesButton = document.getElementById("namesButton");
      const linksButton = document.getElementById("linksButton");
      const speedSlider = document.getElementById("speedSlider");
      const speedValue = document.getElementById("speedValue");
      const readout = document.getElementById("readout");
      const toast = document.getElementById("toast");

      const linksPanel = document.getElementById("linksPanel");
      const linksList = document.getElementById("linksList");

      const LOCAL_STORAGE_KEY = "zodiacLinks.v1";

      const state = {
        warp: false,
        showNames: true,
        linksPanelOpen: false,
        speedPercent: Number(speedSlider.value) || 35,

        pointerTargetX: 0,
        pointerTargetY: 0,
        pointerSmoothedX: 0,
        pointerSmoothedY: 0,

        prefersReducedMotion: window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches,

        // Interactivity
        hoveredInstanceId: null,
        // URL mapping: key -> url
        zodiacLinks: Object.create(null),

        // Toast timer
        toastTimeoutHandle: null
      };

      function setWarpEnabled(enabled) {
        state.warp = Boolean(enabled);
        warpButton.setAttribute("aria-pressed", String(state.warp));
        warpButton.textContent = state.warp ? "ON" : "OFF";
        updateReadout();
      }

      function setShowNames(enabled) {
        state.showNames = Boolean(enabled);
        namesButton.setAttribute("aria-pressed", String(state.showNames));
        namesButton.textContent = state.showNames ? "ON" : "OFF";
      }

      function setLinksPanelOpen(enabled) {
        state.linksPanelOpen = Boolean(enabled);
        linksButton.setAttribute("aria-pressed", String(state.linksPanelOpen));
        linksButton.textContent = state.linksPanelOpen ? "OPEN" : "CLOSED";
        linksPanel.classList.toggle("open", state.linksPanelOpen);
      }

      function setSpeedPercent(value) {
        state.speedPercent = Math.max(0, Math.min(100, Number(value) || 0));
        speedSlider.value = String(state.speedPercent);
        speedValue.textContent = state.speedPercent + "%";
      }

      function updateReadout() {
        if (!readout) return;
        readout.innerHTML =
          "OUTER VIEW: ACTIVE<br />" +
          "PARALLAX: ENABLED<br />" +
          "CONSTELLATIONS: DRIFTING<br />" +
          "WARP: " + (state.warp ? "ENGAGED" : "IDLE");
      }

      function showToast(htmlText, durationMs) {
        const ms = Math.max(900, Number(durationMs) || 1800);
        toast.innerHTML = htmlText;
        toast.classList.add("show");
        if (state.toastTimeoutHandle) window.clearTimeout(state.toastTimeoutHandle);
        state.toastTimeoutHandle = window.setTimeout(() => {
          toast.classList.remove("show");
        }, ms);
      }

      warpButton.addEventListener("click", () => setWarpEnabled(!state.warp));
      namesButton.addEventListener("click", () => setShowNames(!state.showNames));
      linksButton.addEventListener("click", () => setLinksPanelOpen(!state.linksPanelOpen));
      speedSlider.addEventListener("input", (event) => setSpeedPercent(event.target.value));

      // Keyboard shortcuts

      function resizeCanvasToDisplaySize() {
        const bounds = canvas.getBoundingClientRect();
        const dpr = Math.max(1, Math.min(2.5, window.devicePixelRatio || 1));
        const displayWidth = Math.max(1, Math.floor(bounds.width));
        const displayHeight = Math.max(1, Math.floor(bounds.height));

        const neededWidth = Math.floor(displayWidth * dpr);
        const neededHeight = Math.floor(displayHeight * dpr);

        if (canvas.width !== neededWidth || canvas.height !== neededHeight || render.devicePixelRatio !== dpr) {
          canvas.width = neededWidth;
          canvas.height = neededHeight;
          render.devicePixelRatio = dpr;
        }

        render.width = canvas.width;
        render.height = canvas.height;
        render.centerX = render.width * 0.5;
        render.centerY = render.height * 0.5;
        render.radius = Math.min(render.width, render.height) * 0.48;
      }

      function clamp(value, min, max) {
        return Math.max(min, Math.min(max, value));
      }

      function lerp(a, b, t) {
        return a + (b - a) * t;
      }

      function smoothstep(edge0, edge1, x) {
        const t = clamp((x - edge0) / (edge1 - edge0), 0, 1);
        return t * t * (3 - 2 * t);
      }

      function hashToUnitFloat(seed) {
        let x = seed | 0;
        x ^= x << 13;
        x ^= x >> 17;
        x ^= x << 5;
        return ((x >>> 0) % 1000000) / 1000000;
      }

      function colorWithAlpha(hex, alpha) {
        const r = parseInt(hex.slice(1, 3), 16);
        const g = parseInt(hex.slice(3, 5), 16);
        const b = parseInt(hex.slice(5, 7), 16);
        return "rgba(" + r + "," + g + "," + b + "," + alpha + ")";
      }

      function sanitizeUrl(urlRaw) {
        const url = String(urlRaw || "").trim();
        if (!url) return "";
        // Allow http(s) and mailto; otherwise keep but treat as empty (safety baseline)
        if (/^(https?:\/\/)/i.test(url)) return url;
        if (/^mailto:/i.test(url)) return url;
        // Support user typing without scheme: treat as https
        if (/^[a-z0-9.-]+\.[a-z]{2,}(\/.*)?$/i.test(url)) return "https://" + url;
        return "";
      }

      function escapeHtml(text) {
        return String(text).replace(/[&<>"']/g, (c) => ({
          "&": "&amp;",
          "<": "&lt;",
          ">": "&gt;",
          "\"": "&quot;",
          "'": "&#039;"
        }[c]));
      }

      function loadLinksFromStorage() {
        try {
          const raw = localStorage.getItem(LOCAL_STORAGE_KEY);
          if (!raw) return Object.create(null);
          const parsed = JSON.parse(raw);
          const map = Object.create(null);
          for (const entry of zodiacCatalog) {
            const v = parsed && typeof parsed === "object" ? parsed[entry.key] : "";
            const sanitized = sanitizeUrl(v);
            if (sanitized) map[entry.key] = sanitized;
          }
          return map;
        } catch (e) {
          return Object.create(null);
        }
      }

      function saveLinksToStorage() {
        const payload = Object.create(null);
        for (const entry of zodiacCatalog) {
          payload[entry.key] = state.zodiacLinks[entry.key] || "";
        }
        try {
          localStorage.setItem(LOCAL_STORAGE_KEY, JSON.stringify(payload));
        } catch (e) {
          // ignore
        }
      }

      function spawnConstellationInstance(nowSeconds) {
        const index = Math.floor(hashToUnitFloat(++constellationSeed * 77 + 19) * zodiacCatalog.length);
        const entry = zodiacCatalog[index];

        const angle = hashToUnitFloat(++constellationSeed * 91 + 3) * Math.PI * 2;
        const spawnRadius = 1.18;
        const startX = Math.cos(angle) * spawnRadius;
        const startY = Math.sin(angle) * spawnRadius;

        const driftAngle = angle + Math.PI + (hashToUnitFloat(++constellationSeed * 33 + 17) * 0.65 - 0.325);
        const driftSpeed = 0.07 + 0.10 * hashToUnitFloat(++constellationSeed * 55 + 7);

        const scale = constellationConfig.sizeMin +
          (constellationConfig.sizeMax - constellationConfig.sizeMin) * hashToUnitFloat(++constellationSeed * 21 + 13);

        const rotation = (hashToUnitFloat(++constellationSeed * 41 + 23) * 2 - 1) * 0.55;

        const life = constellationConfig.baseLifeSeconds *
          (0.85 + 0.45 * hashToUnitFloat(++constellationSeed * 27 + 5));

        constellationInstances.push({
          id: instanceIdCounter++,
          entry,
          birthTime: nowSeconds,
          lifeSeconds: life,
          x: startX,
          y: startY,
          vx: Math.cos(driftAngle) * driftSpeed,
          vy: Math.sin(driftAngle) * driftSpeed,
          scale,
          rotation,
          flickerPhase: hashToUnitFloat(++constellationSeed * 111 + 9) * Math.PI * 2
        });
      }

      function updateConstellations(deltaSeconds, nowSeconds) {
        timeSinceLastSpawn += deltaSeconds;

        const desiredInterval = state.warp
          ? constellationConfig.spawnIntervalSeconds * 0.65
          : constellationConfig.spawnIntervalSeconds;

        while (timeSinceLastSpawn >= desiredInterval && constellationInstances.length < constellationConfig.maxInstances) {
          timeSinceLastSpawn -= desiredInterval;
          spawnConstellationInstance(nowSeconds);
        }

        for (let i = constellationInstances.length - 1; i >= 0; i--) {
          const inst = constellationInstances[i];
          inst.x += inst.vx * deltaSeconds;
          inst.y += inst.vy * deltaSeconds;

          const age = nowSeconds - inst.birthTime;
          if (age > inst.lifeSeconds) {
            // clear hover if needed
            if (state.hoveredInstanceId === inst.id) state.hoveredInstanceId = null;
            constellationInstances.splice(i, 1);
          }
        }

        if (constellationInstances.length === 0 && nowSeconds > 0.2) {
          spawnConstellationInstance(nowSeconds);
        }
      }

      function drawStars(nowSeconds, cameraOffsetX, cameraOffsetY, baseSpeedPixelsPerSecond, warpIntensity) {
        const time = nowSeconds;

        for (let i = 0; i < stars.length; i++) {
          const star = stars[i];
          const layer = starLayers[star.layerIndex];

          const parallax = layer.depth;
          const camX = cameraOffsetX * parallax;
          const camY = cameraOffsetY * parallax;

          const speed = baseSpeedPixelsPerSecond * layer.speedFactor;
          const normPerSecond = speed / Math.max(1, render.radius * 2.2);

          const driftX = (0.08 + 0.10 * layer.depth) * normPerSecond;
          const driftY = (0.24 + 0.18 * layer.depth) * normPerSecond;

          let x = star.x + time * driftX;
          let y = star.y + time * driftY;

          x = ((x + 1) % 2) - 1;
          y = ((y + 1) % 2) - 1;

          const px = render.centerX + x * render.radius + camX;
          const py = render.centerY + y * render.radius + camY;

          if (
            px < render.centerX - render.radius * 1.15 || px > render.centerX + render.radius * 1.15 ||
            py < render.centerY - render.radius * 1.15 || py > render.centerY + render.radius * 1.15
          ) continue;

          const twinkle = 0.78 + 0.22 * Math.sin(nowSeconds * (1.4 + layer.depth) + star.twinklePhase);
          const alpha = clamp(star.brightness * twinkle, 0.15, 1.0);

          const tint = 0.06 + 0.08 * layer.depth;
          context.fillStyle = "rgba(" +
            Math.floor(220 + 20 * tint) + "," +
            Math.floor(230 + 10 * tint) + "," +
            Math.floor(255) + "," + (alpha * 0.95) + ")";

          const size = star.size * render.devicePixelRatio;
          const r = Math.max(0.6, size);

          if (warpIntensity > 0.02) {
            const streakLength = r * (3.0 + 22.0 * warpIntensity) * layer.depth;
            const dx = -driftX * render.radius * (30 + 90 * warpIntensity);
            const dy = -driftY * render.radius * (30 + 90 * warpIntensity);

            context.strokeStyle = "rgba(255,255,255," + (alpha * (0.20 + 0.30 * warpIntensity)) + ")";
            context.lineWidth = Math.max(1, r * 0.55);
            context.beginPath();
            context.moveTo(px, py);
            const hyp = Math.max(1, Math.hypot(dx, dy));
            context.lineTo(
              px + dx * 0.02 + (dx / hyp) * streakLength,
              py + dy * 0.02 + (dy / hyp) * streakLength
            );
            context.stroke();

            context.fillStyle = "rgba(255,255,255," + (alpha * (0.70 + 0.20 * warpIntensity)) + ")";
            context.beginPath();
            context.arc(px, py, r * 0.75, 0, Math.PI * 2);
            context.fill();
          } else {
            context.beginPath();
            context.arc(px, py, r * 0.65, 0, Math.PI * 2);
            context.fill();

            context.fillStyle = "rgba(94,255,171," + (alpha * 0.05) + ")";
            context.beginPath();
            context.arc(px, py, r * 2.5, 0, Math.PI * 2);
            context.fill();
          }
        }
      }

      /* ----------------------------
         Constellations drawing + hit cache population
      ---------------------------- */
      function drawConstellationInstance(inst, nowSeconds, cameraOffsetX, cameraOffsetY) {
        const age = nowSeconds - inst.birthTime;
        const life = inst.lifeSeconds;

        const fadeIn = smoothstep(0.0, 1.8, age);
        const fadeOut = 1.0 - smoothstep(life - 2.2, life, age);
        const fade = clamp(fadeIn * fadeOut, 0, 1);

        const flicker = 0.86 + 0.14 * Math.sin(nowSeconds * 2.2 + inst.flickerPhase);
        const alpha = fade * flicker;
        if (alpha <= 0.01) {
          hitCache.delete(inst.id);
          return;
        }

        const px = render.centerX + inst.x * render.radius + cameraOffsetX * 0.42;
        const py = render.centerY + inst.y * render.radius + cameraOffsetY * 0.42;

        const scale = inst.scale * render.radius;
        const rot = inst.rotation;

        const cosR = Math.cos(rot);
        const sinR = Math.sin(rot);

        function transformPoint(point) {
          const x = point[0] * scale;
          const y = point[1] * scale;
          return {
            x: px + x * cosR - y * sinR,
            y: py + x * sinR + y * cosR
          };
        }

        // Hover glow boost if clickable and hovered
        const url = state.zodiacLinks[inst.entry.key] || "";
        const clickable = Boolean(url);
        const hovered = (state.hoveredInstanceId === inst.id);

        const accent = "#5EFFAB";
        const glowBoost = (clickable && hovered) ? 1.25 : 1.0;

        const lineOuter = colorWithAlpha(accent, 0.10 * alpha * glowBoost);
        const lineMid = colorWithAlpha(accent, 0.28 * alpha * glowBoost);
        const lineCore = colorWithAlpha(accent, 0.70 * alpha * (clickable ? 1.04 : 1.0));

        context.lineCap = "round";
        context.lineJoin = "round";

        const baseWidth = Math.max(1, render.radius * 0.0045);
        const glowWidth = baseWidth * (4.2 * glowBoost);

        // Collect transformed points once (also used for hit test radius)
        const transformed = new Array(inst.entry.points.length);
        let sumX = 0, sumY = 0;
        for (let i = 0; i < inst.entry.points.length; i++) {
          const p = transformPoint(inst.entry.points[i]);
          transformed[i] = p;
          sumX += p.x;
          sumY += p.y;
        }

        // Hit radius (bounding circle around transformed points)
        const cx = sumX / transformed.length;
        const cy = sumY / transformed.length;

        let maxD = 0;
        for (let i = 0; i < transformed.length; i++) {
          const dx = transformed[i].x - cx;
          const dy = transformed[i].y - cy;
          const d = Math.hypot(dx, dy);
          if (d > maxD) maxD = d;
        }

        // Add padding to cover label and make clicking easier
        const hitPadding = Math.max(14 * render.devicePixelRatio, render.radius * 0.035);
        const hitRadius = maxD + hitPadding;

        hitCache.set(inst.id, {
          id: inst.id,
          x: cx,
          y: cy,
          r: hitRadius,
          clickable,
          url,
          entryKey: inst.entry.key,
          labelRu: inst.entry.labelRu,
          glyph: inst.entry.glyph,
          alpha
        });

        // Outer glow lines
        context.strokeStyle = lineOuter;
        context.lineWidth = glowWidth;
        context.beginPath();
        for (const edge of inst.entry.edges) {
          const a = transformed[edge[0]];
          const b = transformed[edge[1]];
          context.moveTo(a.x, a.y);
          context.lineTo(b.x, b.y);
        }
        context.stroke();

        // Mid glow lines
        context.strokeStyle = lineMid;
        context.lineWidth = baseWidth * (2.2 * glowBoost);
        context.beginPath();
        for (const edge of inst.entry.edges) {
          const a = transformed[edge[0]];
          const b = transformed[edge[1]];
          context.moveTo(a.x, a.y);
          context.lineTo(b.x, b.y);
        }
        context.stroke();

        // Core lines
        context.strokeStyle = lineCore;
        context.lineWidth = baseWidth * 1.05;
        context.beginPath();
        for (const edge of inst.entry.edges) {
          const a = transformed[edge[0]];
          const b = transformed[edge[1]];
          context.moveTo(a.x, a.y);
          context.lineTo(b.x, b.y);
        }
        context.stroke();

        // Stars at points
        for (let i = 0; i < transformed.length; i++) {
          const p = transformed[i];
          const starR = baseWidth * (1.6 + (i % 3) * 0.55);

          const starGlowAlpha = (clickable ? 0.18 : 0.13) * alpha * (hovered ? 1.25 : 1.0);
          context.fillStyle = colorWithAlpha(accent, starGlowAlpha);
          context.beginPath();
          context.arc(p.x, p.y, starR * 3.2 * (hovered ? 1.10 : 1.0), 0, Math.PI * 2);
          context.fill();

          context.fillStyle = "rgba(240,255,250," + (0.72 * alpha) + ")";
          context.beginPath();
          context.arc(p.x, p.y, starR * 0.95, 0, Math.PI * 2);
          context.fill();
        }

        // Label
        if (state.showNames) {
          const label = inst.entry.glyph + " " + inst.entry.labelRu;

          // label anchor: use normalized center of entry, then transform
          let nSumX = 0, nSumY = 0;
          for (const pt of inst.entry.points) { nSumX += pt[0]; nSumY += pt[1]; }
          const ncx = nSumX / inst.entry.points.length;
          const ncy = nSumY / inst.entry.points.length;

          const labelAnchor = transformPoint([ncx, ncy]);
          const labelOffsetX = (0.18 * scale) * cosR - (-0.18 * scale) * sinR;
          const labelOffsetY = (0.18 * scale) * sinR + (-0.18 * scale) * cosR;

          const tx = labelAnchor.x + labelOffsetX;
          const ty = labelAnchor.y + labelOffsetY;

          const fontSize = Math.max(12, Math.round(render.radius * 0.045));
          context.font = "600 " + fontSize + "px ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, 'Roboto Mono', monospace";
          context.textAlign = "left";
          context.textBaseline = "middle";

          const labelGlowAlpha = (clickable ? 0.16 : 0.12) * alpha * (hovered ? 1.25 : 1.0);
          context.fillStyle = colorWithAlpha("#5EFFAB", labelGlowAlpha);
          context.fillText(label, tx + 1, ty + 1);
          context.fillText(label, tx - 1, ty - 1);

          const labelCoreAlpha = (clickable ? 0.88 : 0.78) * alpha;
          context.fillStyle = "rgba(200,255,228," + labelCoreAlpha + ")";
          context.fillText(label, tx, ty);

          // underline scanline
          const metrics = context.measureText(label);
          const underlineY = ty + Math.max(10, render.radius * 0.018);
          context.strokeStyle = colorWithAlpha("#5EFFAB", (clickable ? 0.32 : 0.25) * alpha);
          context.lineWidth = Math.max(1, render.radius * 0.0028);
          context.beginPath();
          context.moveTo(tx, underlineY);
          context.lineTo(tx + metrics.width * 0.6, underlineY);
          context.stroke();

          // small link indicator if URL exists
          if (clickable) {
            context.font = "700 " + Math.max(11, Math.round(fontSize * 0.78)) + "px ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, 'Roboto Mono', monospace";
            context.fillStyle = "rgba(160,255,220," + (0.62 * alpha) + ")";
            context.fillText("↗", tx + metrics.width + 8, ty);
          }
        }
      }

      function drawConstellations(nowSeconds, cameraOffsetX, cameraOffsetY) {
        hitCache.clear();
        for (let i = 0; i < constellationInstances.length; i++) {
          drawConstellationInstance(constellationInstances[i], nowSeconds, cameraOffsetX, cameraOffsetY);
        }
      }

      /* ----------------------------
         Optional canvas ring glow (very light)
      ---------------------------- */
      function drawCanvasNeonRing(nowSeconds) {
        // Very subtle inner neon on the glass edge (kept cheap)
        const pulse = 0.55 + 0.45 * Math.sin(nowSeconds * 1.1);
        const ringWidth = Math.max(1, render.radius * 0.012);

        context.save();
        context.globalCompositeOperation = "screen";
        context.strokeStyle = "rgba(0,255,200," + (0.06 + 0.03 * pulse) + ")";
        context.lineWidth = ringWidth;
        context.beginPath();
        context.arc(render.centerX, render.centerY, render.radius * 0.995, 0, Math.PI * 2);
        context.stroke();

        context.strokeStyle = "rgba(94,255,171," + (0.045 + 0.02 * pulse) + ")";
        context.lineWidth = ringWidth * 2.1;
        context.beginPath();
        context.arc(render.centerX, render.centerY, render.radius * 0.988, 0, Math.PI * 2);
        context.stroke();

        context.restore();
      }
